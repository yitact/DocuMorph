```dart
class _ScannerHomePageState extends State<ScannerHomePage> {
  File? _imageFile;
  String _extractedText = '';
  bool _isProcessing = false;

  final ImagePicker _picker = ImagePicker();
  final TextRecognizer _textRecognizer =
      TextRecognizer(script: TextRecognitionScript.latin);

  @override
  void dispose() {
    _textRecognizer.close();
    super.dispose();
  }

  Future<void> _pickAndProcessImage(ImageSource source) async {
    try {
      final XFile? pickedFile = await _picker.pickImage(
        source: source,
        maxWidth: 2048,
        maxHeight: 2048,
        imageQuality: 90,
      );

      if (pickedFile == null) return;

      setState(() {
        _imageFile = File(pickedFile.path);
        _isProcessing = true;
        _extractedText = '';
      });

      // Process on-device OCR using Google ML Kit
      final inputImage = InputImage.fromFilePath(pickedFile.path);
      final RecognizedText recognizedText =
          await _textRecognizer.processImage(inputImage);

      setState(() {
        _extractedText = recognizedText.text.trim().isEmpty
            ? 'No readable text detected.'
            : recognizedText.text;
        _isProcessing = false;
      });
    } catch (e) {
      setState(() => _isProcessing = false);
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Error scanning document: $e')),
        );
      }
    }
  }

  void _copyToClipboard() {
    if (_extractedText.isEmpty) return;
    Clipboard.setData(ClipboardData(text: _extractedText));
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Extracted text copied to clipboard!')),
    );
  }

  void _shareDigitalText() {
    if (_extractedText.isEmpty) return;
    Share.share(
      _extractedText,
      subject: 'Digitized Document Content',
    );

              tooltip: 'Copy Text',

                  ? ClipRRect(

                      child: Text(
                        'No document captured yet.\nUse the buttons below to start.',
```

```dart
import 'package:google_mlkit_text_recognition/google_mlkit_text_recognition.dart';
import 'ocr_service.dart';

class MlKitOcrService implements OcrService {
  final TextRecognizer _recognizer =
      TextRecognizer(script: TextRecognitionScript.latin);

  @override
  Future<String> recognizeText(File imageFile) async {
    final inputImage = InputImage.fromFilePath(imageFile.path);
    final RecognizedText recognizedText =
        await _recognizer.processImage(inputImage);
    return recognizedText.text.trim();
  }

  @override
  void dispose() {
    _recognizer.close();

}
```

```dart
import 'package:flutter/material.dart';
import 'package:mobile_scanner/mobile_scanner.dart';

class QrScannerScreen extends StatefulWidget {
  const QrScannerScreen({super.key});

  @override
  State<QrScannerScreen> createState() => _QrScannerScreenState();
}

class _QrScannerScreenState extends State<QrScannerScreen> {
  final MobileScannerController _controller = MobileScannerController(
    formats: const [BarcodeFormat.qrCode, BarcodeFormat.dataMatrix, BarcodeFormat.code128],
    detectionSpeed: DetectionSpeed.noDuplicates,
  );
  bool _isScanned = false;

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  void _onDetect(BarcodeCapture capture) {
    if (_isScanned) return;
    final List<Barcode> barcodes = capture.barcodes;
    for (final barcode in barcodes) {
      if (barcode.rawValue != null && barcode.rawValue!.isNotEmpty) {
        setState(() => _isScanned = true);
        Navigator.of(context).pop(barcode.rawValue);
        break;
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Scan QR / Barcode'),
        backgroundColor: Colors.black,
        foregroundColor: Colors.white,
        actions: [
          IconButton(
            icon: ValueListenableBuilder(
              valueListenable: _controller,
              builder: (context, state, child) {
                return Icon(
                  state.torchState == TorchState.on
                      ? Icons.flash_on
                      : Icons.flash_off,
                );
              },
            ),
            onPressed: () => _controller.toggleTorch(),
          ),
          IconButton(
            icon: const Icon(Icons.cameraswitch_outlined),
            onPressed: () => _controller.switchCamera(),
          ),
        ],
      ),
      backgroundColor: Colors.black,
      body: Stack(
        children: [
          MobileScanner(
            controller: _controller,
            onDetect: _onDetect,
          ),
          // Viewfinder reticle overlay
          Center(
            child: Container(
              width: 250,
              height: 250,
              decoration: BoxDecoration(
                border: Border.all(color: const Color(0xFF06B6D4), width: 3),
                borderRadius: BorderRadius.circular(16),

          ),
          const Positioned(
            bottom: 40,
```

**Step 2: Create Scanned Page Model & PDF Generator (`lib/services/pdf_export_service.dart`)**

This service collates captured page images alongside their extracted OCR text into a multi-page PDF document:

```dart
import 'dart:io';
import 'package:pdf/pdf.dart';
import 'package:pdf/widgets.dart' as pw;
import 'package:path_provider/path_provider.dart';

class ScannedDocumentPage {
  final File imageFile;
  final String extractedText;

  ScannedDocumentPage({
    required this.imageFile,
    required this.extractedText,
  });
}

class PdfExportService {
  static Future<File> generateMultiPagePdf({
    required List<ScannedDocumentPage> pages,
    required String documentTitle,
  }) async {
    final pdf = pw.Document(
      title: documentTitle,
      author: 'Samuel (DocuMorph)',
    );

    for (int i = 0; i < pages.length; i++) {
      final page = pages[i];
      final imageBytes = await page.imageFile.readAsBytes();
      final pdfImage = pw.MemoryImage(imageBytes);

      pdf.addPage(
        pw.Page(
          pageFormat: PdfPageFormat.a4,
          margin: const pw.EdgeInsets.all(32),
          build: (pw.Context context) {
            return pw.Column(
              crossAxisAlignment: pw.CrossAxisAlignment.start,
              children: [
                // Header
                pw.Row(
                  mainAxisAlignment: pw.MainAxisAlignment.spaceBetween,
                  children: [
                    pw.Text(
                      documentTitle,
                      style: pw.TextStyle(
                        fontSize: 12,
                        fontWeight: pw.FontWeight.bold,
                        color: PdfColors.grey700,
                      ),
                    ),
                    pw.Text(
                      'Page ${i + 1} of ${pages.length}',
                      style: const pw.TextStyle(fontSize: 10, color: PdfColors.grey600),
                    ),
                  ],
                ),
                pw.Divider(thickness: 0.5, color: PdfColors.grey400),
                pw.SizedBox(height: 12),

                // Scanned Image Preview
                pw.Container(
                  height: 320,
                  alignment: pw.Alignment.center,
                  child: pw.Image(pdfImage, fit: pw.BoxFit.contain),
                ),
                pw.SizedBox(height: 16),

                // Extracted Digital Text
                pw.Text(
                  'Extracted Digital Content:',
                  style: pw.TextStyle(
                    fontSize: 11,
                    fontWeight: pw.FontWeight.bold,
                    color: PdfColors.indigo900,
                  ),
                ),
                pw.SizedBox(height: 6),
                pw.Expanded(
                  child: pw.Container(
                    padding: const pw.EdgeInsets.all(8),
                    decoration: pw.BoxDecoration(
                      border: pw.Border.all(color: PdfColors.grey300, width: 0.5),
                      borderRadius: const pw.BorderRadius.all(pw.Radius.circular(4)),
                      color: PdfColors.grey100,
                    ),
                    child: pw.SingleChildScrollView(
                      child: pw.Text(
                        page.extractedText.isNotEmpty
                            ? page.extractedText
                            : '[No readable text detected on this page]',
                        style: const pw.TextStyle(fontSize: 9, lineSpacing: 1.3),
                      ),
                    ),
                  ),
                ),
                pw.SizedBox(height: 8),

                // Footer
                pw.Center(
                  child: pw.Text(
                    'Digitized with DocuMorph • © 2026 Samuel',
                    style: const pw.TextStyle(fontSize: 8, color: PdfColors.grey500),
                  ),
                ),
              ],
            );
          },
        ),
      );
    }

    final outputDir = await getTemporaryDirectory();
    final file = File('${outputDir.path}/DocuMorph_${DateTime.now().millisecondsSinceEpoch}.pdf');
    await file.writeAsBytes(await pdf.save());
    return file;
  }
}
```

**Step 3: Multi-Page Batch Scan Controller (`lib/screens/batch_scanner_sheet.dart`)**

This interactive modal allows users to capture sequential pages, preview thumbnails, delete misaligned shots, and trigger collation:

```dart
import 'dart:io';
import 'package:flutter/material.dart';
import 'package:image_picker/image_picker.dart';
import 'package:share_plus/share_plus.dart';
import '../services/ocr_service.dart';
import '../services/pdf_export_service.dart';

class BatchScannerSheet extends StatefulWidget {
  final OcrService ocrService;

  const BatchScannerSheet({super.key, required this.ocrService});

  static Future<void> show(BuildContext context, OcrService ocrService) {
    return showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (ctx) => BatchScannerSheet(ocrService: ocrService),
    );
  }

  @override
  State<BatchScannerSheet> createState() => _BatchScannerSheetState();
}

class _BatchScannerSheetState extends State<BatchScannerSheet> {
  final List<ScannedDocumentPage> _batchPages = [];
  final ImagePicker _picker = ImagePicker();
  bool _isProcessing = false;
  String _statusText = 'No pages captured yet.';

  Future<void> _capturePage(ImageSource source) async {
    try {
      final XFile? photo = await _picker.pickImage(
        source: source,
        maxWidth: 2048,
        maxHeight: 2048,
        imageQuality: 90,
      );

      if (photo == null) return;

      setState(() {
        _isProcessing = true;
        _statusText = 'Extracting OCR for page ${_batchPages.length + 1}...';
      });

      final imageFile = File(photo.path);
      final text = await widget.ocrService.recognizeText(imageFile);

      setState(() {
        _batchPages.add(ScannedDocumentPage(
          imageFile: imageFile,
          extractedText: text,
        ));
        _isProcessing = false;
        _statusText = '${_batchPages.length} page(s) ready.';
      });
    } catch (e) {
      setState(() => _isProcessing = false);
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Failed to capture page: $e')),
        );
      }
    }
  }

  Future<void> _generateAndSharePdf() async {
    if (_batchPages.isEmpty) return;

    setState(() {
      _isProcessing = true;
      _statusText = 'Assembling multi-page PDF...';
    });

    try {
      final pdfFile = await PdfExportService.generateMultiPagePdf(
        pages: _batchPages,
        documentTitle: 'DocuMorph Scan (${_batchPages.length} Pages)',
      );

      setState(() => _isProcessing = false);

      await Share.shareXFiles(
        [XFile(pdfFile.path)],
        text: 'Colated multi-page PDF via DocuMorph',
      );
    } catch (e) {
      setState(() => _isProcessing = false);
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Error collating PDF: $e')),
        );
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Container(
      height: MediaQuery.of(context).size.height * 0.85,
      decoration: const BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.vertical(top: Radius.circular(24)),
      ),
      padding: const EdgeInsets.all(20),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          // Header Bar
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: [
              Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  const Text(
                    'Batch Scan & Collate',
                    style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
                  ),
                  Text(_statusText, style: const TextStyle(fontSize: 12, color: Colors.grey)),
                ],
              ),
              IconButton(
                icon: const Icon(Icons.close),
                onPressed: () => Navigator.pop(context),
              )
            ],
          ),
          const Divider(height: 24),

          // Captured Pages Thumbnail Grid
          Expanded(
            child: _batchPages.isEmpty
                ? Center(
                    child: Column(
                      mainAxisSize: MainAxisSize.min,
                      children: const [
                        Icon(Icons.photo_library_outlined, size: 48, color: Colors.grey),
                        SizedBox(height: 8),
                        Text(
                          'Add multiple pages sequentially\nusing the camera or gallery below.',
                          textAlign: TextAlign.center,
                          style: TextStyle(color: Colors.grey),
                        ),
                      ],
                    ),
                  )
                : GridView.builder(
                    gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
                      crossAxisCount: 2,
                      crossAxisSpacing: 12,
                      mainAxisSpacing: 12,
                      childAspectRatio: 0.75,
                    ),
                    itemCount: _batchPages.length,
                    itemBuilder: (ctx, index) {
                      final item = _batchPages[index];
                      return Stack(
                        children: [
                          Container(
                            decoration: BoxDecoration(
                              borderRadius: BorderRadius.circular(12),
                              border: Border.all(color: Colors.grey.shade300),
                              image: DecorationImage(
                                image: FileImage(item.imageFile),
                                fit: BoxFit.cover,
                              ),
                            ),
                          ),
                          Positioned(
                            top: 6,
                            left: 6,
                            child: CircleAvatar(
                              radius: 12,
                              backgroundColor: const Color(0xFF4F46E5),
                              child: Text(
                                '${index + 1}',
                                style: const TextStyle(color: Colors.white, fontSize: 11),
                              ),
                            ),
                          ),
                          Positioned(
                            top: 6,
                            right: 6,
                            child: InkWell(
                              onTap: () => setState(() => _batchPages.removeAt(index)),
                              child: const CircleAvatar(
                                radius: 12,
                                backgroundColor: Colors.black54,
                                child: Icon(Icons.close, size: 14, color: Colors.white),
                              ),

            const Padding(
```

**Step 2: Define Data Models & Parsers (`lib/services/structured_parser_service.dart`)**

This file provides a unified data structure, an offline rule-based regex parser (for dates, totals, taxes, and vendor names), and an AI fallback/enhancement pipeline.

```dart
import 'dart:convert';
import 'dart:io';
import 'package:csv/csv.dart';
import 'package:http/http.dart' as http;

class LineItem {
  final String description;
  final double price;
  final int quantity;

  LineItem({
    required this.description,
    required this.price,
    this.quantity = 1,
  });

  Map<String, dynamic> toJson() => {
        'description': description,
        'quantity': quantity,
        'price': price,
      };

  factory LineItem.fromJson(Map<String, dynamic> json) => LineItem(
        description: json['description'] ?? 'Item',
        price: (json['price'] as num?)?.toDouble() ?? 0.0,
        quantity: (json['quantity'] as num?)?.toInt() ?? 1,
      );
}

class ParsedDocument {
  final String vendor;
  final String date;
  final String invoiceNumber;
  final double subtotal;
  final double tax;
  final double total;
  final List<LineItem> items;

  ParsedDocument({
    required this.vendor,
    required this.date,
    required this.invoiceNumber,
    required this.subtotal,
    required this.tax,
    required this.total,
    required this.items,
  });

  Map<String, dynamic> toJson() => {
        'vendor': vendor,
        'date': date,
        'invoiceNumber': invoiceNumber,
        'subtotal': subtotal,
        'tax': tax,
        'total': total,
        'items': items.map((e) => e.toJson()).toList(),
      };

  String toPrettyJson() => const JsonEncoder.withIndent('  ').convert(toJson());

  String toCsv() {
    List<List<dynamic>> rows = [
      ['Vendor', 'Invoice #', 'Date', 'Subtotal', 'Tax', 'Total'],
      [vendor, invoiceNumber, date, subtotal, tax, total],
      [],
      ['Item Description', 'Quantity', 'Price (USD)']
    ];

    for (var item in items) {
      rows.add([item.description, item.quantity, item.price]);
    }
    return const ListToCsvConverter().convert(rows);
  }
}

class StructuredParserService {
  // -------------------------------------------------------------
  // 1. FAST ON-DEVICE REGEX PARSER (Zero Cloud / Offline)
  // -------------------------------------------------------------
  static ParsedDocument parseWithRegex(String rawText) {
    final lines = rawText.split('\n').map((l) => l.trim()).where((l) => l.isNotEmpty).toList();

    String vendor = lines.isNotEmpty ? lines.first : 'Unknown Vendor';
    String invoiceNumber = '';
    String date = '';
    double total = 0.0;
    double tax = 0.0;
    double subtotal = 0.0;
    List<LineItem> detectedItems = [];

    // Date regex (MM/DD/YYYY, YYYY-MM-DD, DD-MM-YYYY)
    final datePattern = RegExp(r'\b(?:\d{1,2}[/-]\d{1,2}[/-]\d{2,4}|\d{4}[/-]\d{1,2}[/-]\d{1,2})\b');
    final invoicePattern = RegExp(r'(?:invoice|inv|receipt|bill)\s*(?:#|no\.?|num)?[:\s]*([A-Z0-9-]+)', caseSensitive: false);
    final totalPattern = RegExp(r'(?:grand\s*total|total|amount\s*due|balance\s*due)\D*([0-9]+\.[0-9]{2})', caseSensitive: false);
    final taxPattern = RegExp(r'(?:tax|vat|hst|gst)\D*([0-9]+\.[0-9]{2})', caseSensitive: false);
    final subtotalPattern = RegExp(r'(?:subtotal|sub-total|sub\s*total)\D*([0-9]+\.[0-9]{2})', caseSensitive: false);
    final lineItemPattern = RegExp(r'^(.*?)\s+([0-9]+\.[0-9]{2})$');

    for (var line in lines) {
      if (date.isEmpty) {
        final match = datePattern.firstMatch(line);
        if (match != null) date = match.group(0) ?? '';
      }

      if (invoiceNumber.isEmpty) {
        final match = invoicePattern.firstMatch(line);
        if (match != null) invoiceNumber = match.group(1) ?? '';
      }

      final totalMatch = totalPattern.firstMatch(line);
      if (totalMatch != null) {
        total = double.tryParse(totalMatch.group(1) ?? '') ?? total;
      }

      final taxMatch = taxPattern.firstMatch(line);
      if (taxMatch != null) {
        tax = double.tryParse(taxMatch.group(1) ?? '') ?? tax;
      }

      final subMatch = subtotalPattern.firstMatch(line);
      if (subMatch != null) {
        subtotal = double.tryParse(subMatch.group(1) ?? '') ?? subtotal;
      }

      // Line item match
      final itemMatch = lineItemPattern.firstMatch(line);
      if (itemMatch != null && !line.toLowerCase().contains('total') && !line.toLowerCase().contains('tax')) {
        final desc = itemMatch.group(1)?.trim() ?? 'Item';
        final price = double.tryParse(itemMatch.group(2) ?? '') ?? 0.0;
        if (price > 0) {
          detectedItems.add(LineItem(description: desc, price: price));
        }
      }
    }

    if (total == 0.0 && subtotal > 0) {
      total = subtotal + tax;
    }

    return ParsedDocument(
      vendor: vendor,
      date: date.isNotEmpty ? date : DateTime.now().toIso8601String().split('T').first,
      invoiceNumber: invoiceNumber.isNotEmpty ? invoiceNumber : 'N/A',
      subtotal: subtotal,
      tax: tax,
      total: total,
      items: detectedItems,
    );
  }

  // -------------------------------------------------------------
  // 2. AI MULTIMODAL STRUCTURED PARSER (Gemini Vision / Cloud)
  // -------------------------------------------------------------
  static Future<ParsedDocument> parseWithAi({
    required File imageFile,
    required String apiKey,
  }) async {
    final imageBytes = await imageFile.readAsBytes();
    final base64Image = base64Encode(imageBytes);

    final url = Uri.parse(
        'https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=$apiKey');

    final prompt = '''
Extract the structured fields from this receipt/invoice. Return ONLY valid JSON with no markdown formatting:
{
  "vendor": "string",
  "date": "YYYY-MM-DD",
  "invoiceNumber": "string",
  "subtotal": 0.00,
  "tax": 0.00,
  "total": 0.00,
  "items": [
    {"description": "string", "quantity": 1, "price": 0.00}
  ]
}
''';

    final requestBody = jsonEncode({
      "contents": [
        {
          "parts": [
            {"text": prompt},
            {
              "inline_data": {
                "mime_type": "image/jpeg",
                "data": base64Image,
              }
            }
          ]
        }
      ],
      "generationConfig": {
        "response_mime_type": "application/json",
        "temperature": 0.1
      }
    });

    final response = await http.post(
      url,
      headers: {'Content-Type': 'application/json'},
      body: requestBody,
    );

    if (response.statusCode != 200) {
      throw Exception('AI Parsing failed with status ${response.statusCode}: ${response.body}');
    }

    final data = jsonDecode(response.body);
    final String contentText =
        data['candidates'][0]['content']['parts'][0]['text'];
    final Map<String, dynamic> parsedJson = jsonDecode(contentText);

    List<LineItem> itemsList = [];
    if (parsedJson['items'] != null) {
      for (var it in parsedJson['items']) {
        itemsList.add(LineItem.fromJson(it));
      }
    }

    return ParsedDocument(
      vendor: parsedJson['vendor'] ?? 'Unknown Vendor',
      date: parsedJson['date'] ?? '',
      invoiceNumber: parsedJson['invoiceNumber'] ?? 'N/A',
      subtotal: (parsedJson['subtotal'] as num?)?.toDouble() ?? 0.0,
      tax: (parsedJson['tax'] as num?)?.toDouble() ?? 0.0,
      total: (parsedJson['total'] as num?)?.toDouble() ?? 0.0,
      items: itemsList,
    );
  }
}
```

### Step 2: Create Data Models & Storage Service (`lib/services/storage_service.dart`)

This service manages initialization, saving, deleting, and full-text searching across all stored documents and receipts.

```dart
import 'dart:convert';
import 'package:hive_flutter/hive_flutter.dart';
import 'structured_parser_service.dart';

class StoredDocument {
  final String id;
  final String title;
  final String rawText;
  final String category; // 'Receipt', 'Invoice', 'General'
  final DateTime createdAt;
  final String? imagePath;
  final ParsedDocument? structuredData;

  StoredDocument({
    required this.id,
    required this.title,
    required this.rawText,
    required this.category,
    required this.createdAt,
    this.imagePath,
    this.structuredData,
  });

  Map<String, dynamic> toMap() => {
        'id': id,
        'title': title,
        'rawText': rawText,
        'category': category,
        'createdAt': createdAt.toIso8601String(),
        'imagePath': imagePath,
        'structuredData': structuredData != null ? jsonEncode(structuredData!.toJson()) : null,
      };

  factory StoredDocument.fromMap(Map<dynamic, dynamic> map) {
    ParsedDocument? parsed;
    if (map['structuredData'] != null) {
      try {
        final decoded = jsonDecode(map['structuredData'] as String);
        List<LineItem> itemsList = [];
        if (decoded['items'] != null) {
          for (var item in decoded['items']) {
            itemsList.add(LineItem.fromJson(item));
          }
        }
        parsed = ParsedDocument(
          vendor: decoded['vendor'] ?? '',
          date: decoded['date'] ?? '',
          invoiceNumber: decoded['invoiceNumber'] ?? '',
          subtotal: (decoded['subtotal'] as num?)?.toDouble() ?? 0.0,
          tax: (decoded['tax'] as num?)?.toDouble() ?? 0.0,
          total: (decoded['total'] as num?)?.toDouble() ?? 0.0,
          items: itemsList,
        );
      } catch (_) {}
    }

    return StoredDocument(
      id: map['id'] ?? '',
      title: map['title'] ?? 'Untitled Document',
      rawText: map['rawText'] ?? '',
      category: map['category'] ?? 'General',
      createdAt: DateTime.tryParse(map['createdAt'] ?? '') ?? DateTime.now(),
      imagePath: map['imagePath'],
      structuredData: parsed,
    );
  }
}

class StorageService {
  static const String _boxName = 'documents_box';

  static Future<void> init() async {
    await Hive.initFlutter();
    await Hive.openBox(_boxName);
  }

  static Box _getBox() => Hive.box(_boxName);

  static Future<void> saveDocument(StoredDocument doc) async {
    final box = _getBox();
    await box.put(doc.id, doc.toMap());
  }

  static List<StoredDocument> getAllDocuments() {
    final box = _getBox();
    final List<StoredDocument> docs = [];
    for (var key in box.keys) {
      final data = box.get(key);
      if (data is Map) {
        docs.add(StoredDocument.fromMap(data));
      }
    }
    docs.sort((a, b) => b.createdAt.compareTo(a.createdAt));
    return docs;
  }

  static List<StoredDocument> searchDocuments(String query) {
    final all = getAllDocuments();
    if (query.trim().isEmpty) return all;
    final lower = query.toLowerCase();

    return all.where((doc) {
      final inTitle = doc.title.toLowerCase().contains(lower);
      final inText = doc.rawText.toLowerCase().contains(lower);
      final inVendor = doc.structuredData?.vendor.toLowerCase().contains(lower) ?? false;
      return inTitle || inText || inVendor;
    }).toList();
  }

  static Future<void> deleteDocument(String id) async {
    final box = _getBox();
    await box.delete(id);
  }
}
```

### Step 2: Create Backup & Export Service (`lib/services/backup_service.dart`)

This service collects all saved documents from Hive, serializes the metadata/database to JSON, gathers all stored document images, compiles a master summary CSV, and bundles everything into a single timestamped `.zip` file.

```dart
import 'dart:convert';
import 'dart:io';
import 'package:archive/archive_io.dart';
import 'package:csv/csv.dart';
import 'package:path_provider/path_provider.dart';
import 'package:share_plus/share_plus.dart';
import 'storage_service.dart';

class BackupResult {
  final File zipFile;
  final int totalDocuments;
  final int totalImages;
  final int byteSize;

  BackupResult({
    required this.zipFile,
    required this.totalDocuments,
    required this.totalImages,
    required this.byteSize,
  });
}

class BackupService {
  /// Compiles database records, CSV summaries, and image files into a single ZIP archive.
  static Future<BackupResult> createFullBackup() async {
    final List<StoredDocument> allDocs = StorageService.getAllDocuments();
    final Archive archive = Archive();

    // 1. Generate Full Database JSON
    final List<Map<String, dynamic>> dbDump = allDocs.map((doc) => doc.toMap()).toList();
    final String jsonDatabase = const JsonEncoder.withIndent('  ').convert({
      'backupVersion': '1.0.0',
      'app': 'DocuMorph',
      'author': 'Samuel',
      'exportedAt': DateTime.now().toIso8601String(),
      'totalRecords': allDocs.length,
      'documents': dbDump,
    });
    final List<int> jsonBytes = utf8.encode(jsonDatabase);
    archive.addFile(ArchiveFile('database.json', jsonBytes.length, jsonBytes));

    // 2. Generate Master Summary CSV (Spreadsheet friendly)
    List<List<dynamic>> csvRows = [
      ['ID', 'Title', 'Category', 'Created At', 'Vendor', 'Invoice #', 'Date', 'Subtotal', 'Tax', 'Total', 'Raw Text Snippet'],
    ];

    for (final doc in allDocs) {
      final s = doc.structuredData;
      csvRows.add([
        doc.id,
        doc.title,
        doc.category,
        doc.createdAt.toIso8601String(),
        s?.vendor ?? '',
        s?.invoiceNumber ?? '',
        s?.date ?? '',
        s?.subtotal ?? '',
        s?.tax ?? '',
        s?.total ?? '',
        doc.rawText.replaceAll('\n', ' ').trim(),
      ]);
    }

    final String csvContent = const ListToCsvConverter().convert(csvRows);
    final List<int> csvBytes = utf8.encode(csvContent);
    archive.addFile(ArchiveFile('documents_summary.csv', csvBytes.length, csvBytes));

    // 3. Include Images in /images folder inside ZIP
    int imagesIncluded = 0;
    for (final doc in allDocs) {
      if (doc.imagePath != null && doc.imagePath!.isNotEmpty) {
        final imgFile = File(doc.imagePath!);
        if (await imgFile.exists()) {
          final imgBytes = await imgFile.readAsBytes();
          final fileExtension = imgFile.path.split('.').last;
          final archivePath = 'images/${doc.id}.$fileExtension';
          archive.addFile(ArchiveFile(archivePath, imgBytes.length, imgBytes));
          imagesIncluded++;
        }
      }
    }

    // 4. Encode ZIP File
    final ZipEncoder encoder = ZipEncoder();
    final List<int>? encodedZip = encoder.encode(archive);

    if (encodedZip == null) {
      throw Exception('Failed to encode ZIP archive.');
    }

    // 5. Save to Temporary Storage for Sharing
    final tempDir = await getTemporaryDirectory();
    final timestamp = DateTime.now().toIso8601String().replaceAll(':', '-').split('.').first;
    final String zipPath = '${tempDir.path}/DocuMorph_Backup_$timestamp.zip';
    final File zipFile = File(zipPath);
    await zipFile.writeAsBytes(encodedZip);

    return BackupResult(
      zipFile: zipFile,
      totalDocuments: allDocs.length,
      totalImages: imagesIncluded,
      byteSize: encodedZip.length,
    );
  }

  /// Triggers OS Native Share Sheet with the generated ZIP backup.
  static Future<void> exportAndShareBackup() async {
    final result = await createFullBackup();
    await Share.shareXFiles(
      [XFile(result.zipFile.path)],
      subject: 'DocuMorph Database Backup (${result.totalDocuments} items)',
      text: 'DocuMorph Full Backup — Includes database.json, summary CSV, and ${result.totalImages} original scanned images.',
    );
  }
}
```

### Step 2: Implement the Restore Engine (`lib/services/restore_service.dart`)

This service unzips the archive, saves the bundled images into the app's persistent `ApplicationDocumentsDirectory`, reads and validates `database.json`, remaps the stored image paths to the current device's storage directory, and batch-inserts the restored records into Hive.

```dart
import 'dart:convert';
import 'dart:io';
import 'package:archive/archive_io.dart';
import 'package:file_picker/file_picker.dart';
import 'package:path_provider/path_provider.dart';
import 'storage_service.dart';

class RestoreSummary {
  final int totalImported;
  final int imagesRestored;
  final String backupDate;

  RestoreSummary({
    required this.totalImported,
    required this.imagesRestored,
    required this.backupDate,
  });
}

class RestoreService {
  /// Prompts user for a ZIP file and restores database records & image files.
  static Future<RestoreSummary?> pickAndRestoreZip({bool overwriteExisting = false}) async {
    // 1. Pick ZIP file from device
    final FilePickerResult? picked = await FilePicker.platform.pickFiles(
      type: FileType.custom,
      allowedExtensions: ['zip'],
    );

    if (picked == null || picked.files.single.path == null) {
      return null; // User cancelled
    }

    final File zipFile = File(picked.files.single.path!);
    final List<int> bytes = await zipFile.readAsBytes();

    // 2. Decode ZIP archive
    final Archive archive = ZipDecoder().decodeBytes(bytes);

    // 3. Locate target persistent image directory
    final appDocDir = await getApplicationDocumentsDirectory();
    final imagesDir = Directory('${appDocDir.path}/documorph_images');
    if (!await imagesDir.exists()) {
      await imagesDir.create(recursive: true);
    }

    String? jsonContent;
    final Map<String, String> restoredImageMap = {}; // Maps doc ID to new local path
    int imagesRestoredCount = 0;

    // 4. Extract files from archive
    for (final ArchiveFile file in archive) {
      if (file.isFile) {
        if (file.name == 'database.json') {
          jsonContent = utf8.decode(file.content as List<int>);
        } else if (file.name.startsWith('images/')) {
          final String fileName = file.name.replaceFirst('images/', '');
          if (fileName.isNotEmpty) {
            final File diskImage = File('${imagesDir.path}/$fileName');
            await diskImage.writeAsBytes(file.content as List<int>);
            
            // Extract document ID from filename (e.g. "1724538120000.jpg" -> "1724538120000")
            final String docId = fileName.split('.').first;
            restoredImageMap[docId] = diskImage.path;
            imagesRestoredCount++;
          }
        }
      }
    }

    if (jsonContent == null) {
      throw Exception('Invalid backup: Missing database.json in the ZIP archive.');
    }

    // 5. Parse JSON and update Hive
    final Map<String, dynamic> parsedData = jsonDecode(jsonContent);
    final List<dynamic> documentsJson = parsedData['documents'] ?? [];
    final String backupDate = parsedData['exportedAt'] ?? 'Unknown';

    int importedCount = 0;

    for (final docData in documentsJson) {
      if (docData is Map) {
        final Map<String, dynamic> mutableMap = Map<String, dynamic>.from(docData);
        final String docId = mutableMap['id'] ?? '';

        // Remap image path if image was restored in this package
        if (restoredImageMap.containsKey(docId)) {
          mutableMap['imagePath'] = restoredImageMap[docId];
        }

        final StoredDocument doc = StoredDocument.fromMap(mutableMap);
        await StorageService.saveDocument(doc);
        importedCount++;
      }
    }

    return RestoreSummary(
      totalImported: importedCount,
      imagesRestored: imagesRestoredCount,
      backupDate: backupDate,
    );
  }
}
```

### Step 2: Create the Expense Analytics Aggregator (`lib/services/analytics_service.dart`)

This service reads saved documents from Hive, filters for receipt/invoice items, and groups spending by month and vendor:

```dart
import 'storage_service.dart';

class MonthlySpending {
  final int year;
  final int month;
  final double totalAmount;
  final int transactionCount;

  MonthlySpending({
    required this.year,
    required this.month,
    required this.totalAmount,
    required this.transactionCount,
  });

  String get monthName {
    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
    return months[month - 1];
  }
}

class VendorBreakdown {
  final String vendor;
  final double totalAmount;
  final double percentage;

  VendorBreakdown({
    required this.vendor,
    required this.totalAmount,
    required this.percentage,
  });
}

class AnalyticsSummary {
  final double totalSpentAllTime;
  final double currentMonthTotal;
  final double averageTicketSize;
  final List<MonthlySpending> monthlyTotals;
  final List<VendorBreakdown> topVendors;

  AnalyticsSummary({
    required this.totalSpentAllTime,
    required this.currentMonthTotal,
    required this.averageTicketSize,
    required this.monthlyTotals,
    required this.topVendors,
  });
}

class AnalyticsService {
  static AnalyticsSummary calculateMetrics() {
    final List<StoredDocument> allDocs = StorageService.getAllDocuments();
    final List<StoredDocument> receipts = allDocs.where((d) => d.structuredData != null).toList();

    double allTimeTotal = 0.0;
    final Map<String, double> monthlyMap = {};
    final Map<String, int> monthlyCount = {};
    final Map<String, double> vendorMap = {};

    final now = DateTime.now();
    final currentMonthKey = '${now.year}-${now.month.toString().padLeft(2, '0')}';

    for (final doc in receipts) {
      final s = doc.structuredData!;
      final amount = s.total;
      allTimeTotal += amount;

      // Extract date
      DateTime docDate = doc.createdAt;
      if (s.date.isNotEmpty) {
        final parsed = DateTime.tryParse(s.date);
        if (parsed != null) docDate = parsed;
      }

      final monthKey = '${docDate.year}-${docDate.month.toString().padLeft(2, '0')}';
      monthlyMap[monthKey] = (monthlyMap[monthKey] ?? 0.0) + amount;
      monthlyCount[monthKey] = (monthlyCount[monthKey] ?? 0) + 1;

      final vendor = s.vendor.trim().isEmpty ? 'Unknown' : s.vendor.trim();
      vendorMap[vendor] = (vendorMap[vendor] ?? 0.0) + amount;
    }

    // Sort last 6 months chronologically
    final sortedMonthKeys = monthlyMap.keys.toList()..sort();
    final recentKeys = sortedMonthKeys.length > 6
        ? sortedMonthKeys.sublist(sortedMonthKeys.length - 6)
        : sortedMonthKeys;

    final List<MonthlySpending> monthlyList = recentKeys.map((key) {
      final parts = key.split('-');
      return MonthlySpending(
        year: int.parse(parts[0]),
        month: int.parse(parts[1]),
        totalAmount: monthlyMap[key] ?? 0.0,
        transactionCount: monthlyCount[key] ?? 0,
      );
    }).toList();

    // Top 5 Vendors
    final sortedVendors = vendorMap.entries.toList()
      ..sort((a, b) => b.value.compareTo(a.value));
    
    final top5 = sortedVendors.take(5);
    final List<VendorBreakdown> vendorList = top5.map((entry) {
      return VendorBreakdown(
        vendor: entry.key,
        totalAmount: entry.value,
        percentage: allTimeTotal > 0 ? (entry.value / allTimeTotal) * 100 : 0.0,
      );
    }).toList();

    return AnalyticsSummary(
      totalSpentAllTime: allTimeTotal,
      currentMonthTotal: monthlyMap[currentMonthKey] ?? 0.0,
      averageTicketSize: receipts.isNotEmpty ? allTimeTotal / receipts.length : 0.0,
      monthlyTotals: monthlyList,
      topVendors: vendorList,
    );
  }
}
```

### Step 1: Define Expense Categories & Taxonomy Rules (`lib/services/categorization_service.dart`)

This service scores the extracted vendor name, raw OCR text, and line items against categorized keyword lists to assign the best matching category, complete with UI icons and color themes:

```dart
import 'package:flutter/material.dart';
import 'structured_parser_service.dart';

enum ExpenseCategory {
  meals('Meals & Dining', Icons.restaurant, Color(0xFFF59E0B)),
  groceries('Groceries', Icons.shopping_cart, Color(0xFF10B981)),
  travel('Travel & Transport', Icons.directions_car, Color(0xFF06B6D4)),
  office('Office & Tech', Icons.computer, Color(0xFF6366F1)),
  utilities('Utilities & Bills', Icons.bolt, Color(0xFFEC4899)),
  healthcare('Healthcare & Pharmacy', Icons.local_pharmacy, Color(0xFFEF4444)),
  entertainment('Entertainment', Icons.movie, Color(0xFF8B5CF6)),
  general('General / Other', Icons.receipt_long, Color(0xFF64748B));

  final String label;
  final IconData icon;
  final Color color;

  const ExpenseCategory(this.label, this.icon, this.color);

  static ExpenseCategory fromString(String? name) {
    if (name == null) return ExpenseCategory.general;
    return ExpenseCategory.values.firstWhere(
      (e) => e.name.toLowerCase() == name.toLowerCase() || e.label.toLowerCase() == name.toLowerCase(),
      orElse: () => ExpenseCategory.general,
    );
  }
}

class CategorizationService {
  static final Map<ExpenseCategory, List<String>> _keywords = {
    ExpenseCategory.meals: [
      'restaurant', 'cafe', 'coffee', 'bakery', 'starbucks', 'dunkin', 'mcdonald',
      'burger', 'pizza', 'chipotle', 'subway', 'bar', 'grill', 'diner', 'bistro',
      'taco', 'sushi', 'doordash', 'ubereats', 'grubhub', 'steak', 'breakfast', 'lunch', 'dinner'
    ],
    ExpenseCategory.groceries: [
      'market', 'supermarket', 'walmart', 'kroger', 'costco', 'target', 'trader joe',
      'whole foods', 'safeway', 'aldi', 'grocer', 'produce', 'deli', 'bakery', 'food lion', 'publix'
    ],
    ExpenseCategory.travel: [
      'uber', 'lyft', 'taxi', 'cab', 'flight', 'airline', 'delta', 'united', 'american air',
      'hotel', 'motel', 'airbnb', 'parking', 'toll', 'chevron', 'shell', 'exxon', 'bp',
      'gas', 'fuel', 'speedway', 'metro', 'transit', 'train', 'amtrak', 'car rental', 'hertz'
    ],
    ExpenseCategory.office: [
      'apple', 'google', 'microsoft', 'amazon', 'best buy', 'staples', 'office depot',
      'software', 'github', 'adobe', 'hosting', 'aws', 'domain', 'cloud', 'hardware',
      'laptop', 'printer', 'ink', 'paper', 'electronics'
    ],
    ExpenseCategory.utilities: [
      'electric', 'power', 'energy', 'water', 'sewer', 'gas company', 'internet',
      'verizon', 'att', 'at&t', 't-mobile', 'xfinity', 'comcast', 'spectrum', 'trash', 'waste'
    ],
    ExpenseCategory.healthcare: [
      'pharmacy', 'cvs', 'walgreens', 'rite aid', 'clinic', 'hospital', 'doctor',
      'dental', 'optometry', 'vision', 'rx', 'medical', 'prescription', 'health'
    ],
    ExpenseCategory.entertainment: [
      'cinema', 'theatre', 'theater', 'amc', 'regal', 'netflix', 'spotify', 'hulu',
      'disney', 'ticket', 'concert', 'museum', 'bowling', 'arcade', 'steam'
    ],
  };

  /// Evaluates vendor and OCR tokens to detect the most relevant category.
  static ExpenseCategory detectCategory({
    required String vendor,
    required String rawText,
    List<LineItem> items = const [],
  }) {
    final combined = '${vendor.toLowerCase()} ${rawText.toLowerCase()} ${items.map((i) => i.description.toLowerCase()).join(' ')}';
    
    ExpenseCategory bestCategory = ExpenseCategory.general;
    int highestScore = 0;

    _keywords.forEach((category, words) {
      int score = 0;
      for (final word in words) {
        // Vendor match gives double weight
        if (vendor.toLowerCase().contains(word)) {
          score += 3;
        }
        if (combined.contains(word)) {
          score += 1;
        }
      }

      if (score > highestScore) {
        highestScore = score;
        bestCategory = category;
      }
    });

    return bestCategory;
  }
}
```

**Step 1: Create Tax Report Generator Service (`lib/services/tax_report_service.dart`)**

This service aggregates receipts for a selected month, calculates tax write-off subtotals by category, breaks down deductible sales tax/VAT, and outputs a formatted multi-page PDF document:

```dart
import 'dart:io';
import 'package:intl/intl.dart';
import 'package:path_provider/path_provider.dart';
import 'package:pdf/pdf.dart';
import 'package:pdf/widgets.dart' as pw;
import 'package:share_plus/share_plus.dart';
import 'categorization_service.dart';
import 'storage_service.dart';

class TaxReportService {
  static final _currency = NumberFormat.simpleCurrency();

  static Future<File> generateMonthlyTaxPdf({
    required int year,
    required int month,
  }) async {
    final pdf = pw.Document(
      title: 'Tax Expense Report - $month/$year',
      author: 'Samuel (DocuMorph)',
    );

    final List<StoredDocument> allDocs = StorageService.getAllDocuments();
    
    // Filter receipts matching target month & year
    final monthlyReceipts = allDocs.where((doc) {
      if (doc.structuredData == null) return false;
      final s = doc.structuredData!;
      DateTime d = doc.createdAt;
      if (s.date.isNotEmpty) {
        final parsed = DateTime.tryParse(s.date);
        if (parsed != null) d = parsed;
      }
      return d.year == year && d.month == month;
    }).toList();

    // Aggregate totals
    double totalSpend = 0.0;
    double totalTaxPaid = 0.0;
    final Map<ExpenseCategory, double> categoryTotals = {};

    for (final doc in monthlyReceipts) {
      final s = doc.structuredData!;
      totalSpend += s.total;
      totalTaxPaid += s.tax;
      categoryTotals[s.category] = (categoryTotals[s.category] ?? 0.0) + s.total;
    }

    final monthLabel = DateFormat('MMMM yyyy').format(DateTime(year, month));

    pdf.addPage(
      pw.MultiPage(
        pageFormat: PdfPageFormat.a4,
        margin: const pw.EdgeInsets.all(36),
        header: (context) => _buildPdfHeader(monthLabel, monthlyReceipts.length),
        footer: (context) => _buildPdfFooter(context),
        build: (context) => [

                _buildSummaryBlock('TOTAL EXPENSES', _currency.format(totalSpend), PdfColors.indigo900),
                _buildSummaryBlock('SALES TAX / VAT PAID', _currency.format(totalTaxPaid), PdfColors.teal800),
                _buildSummaryBlock('RECEIPTS LOGGED', '${monthlyReceipts.length}', PdfColors.grey800),
              ],
            ),
          ),
          pw.SizedBox(height: 20),

          // Category Breakdown Section
          pw.Text(
            'Schedule C / Category Allocations',
            style: pw.TextStyle(fontSize: 12, fontWeight: pw.FontWeight.bold, color: PdfColors.indigo900),
          ),
          pw.SizedBox(height: 8),
          pw.Table(
            border: pw.TableBorder.all(color: PdfColors.grey300, width: 0.5),
            columnWidths: {
              0: const pw.FlexColumnWidth(3),
              1: const pw.FlexColumnWidth(1),
              2: const pw.FlexColumnWidth(2),
            },
            children: [

                decoration: const pw.BoxDecoration(color: PdfColors.grey200),

                  _tableHeader('Share'),
```

### Step 4: Budget Manager & Threshold Evaluation Service (`lib/services/budget_service.dart`)

This service stores budget limits in Hive and checks monthly spend against these limits each time a receipt is saved:

```dart
import 'package:hive_flutter/hive_flutter.dart';
import 'categorization_service.dart';
import 'notification_service.dart';
import 'storage_service.dart';

class BudgetService {
  static const String _boxName = 'budgets_box';

  static Future<void> init() async {
    await Hive.openBox<double>(_boxName);
  }

  static Box<double> _getBox() => Hive.box<double>(_boxName);

  /// Get budget for a given category (default 0.0 means unconstrained)
  static double getBudget(ExpenseCategory category) {
    return _getBox().get(category.name, defaultValue: 0.0) ?? 0.0;
  }

  /// Save or update a budget limit
  static Future<void> setBudget(ExpenseCategory category, double amount) async {
    await _getBox().put(category.name, amount);
  }

  /// Calculate total spent in the current month for a given category
  static double getCurrentMonthSpent(ExpenseCategory category) {
    final now = DateTime.now();
    final allDocs = StorageService.getAllDocuments();

    double total = 0.0;
    for (final doc in allDocs) {
      if (doc.structuredData != null && doc.structuredData!.category == category) {
        DateTime date = doc.createdAt;
        if (doc.structuredData!.date.isNotEmpty) {
          final parsed = DateTime.tryParse(doc.structuredData!.date);
          if (parsed != null) date = parsed;
        }
        if (date.year == now.year && date.month == now.month) {
          total += doc.structuredData!.total;
        }
      }
    }
    return total;
  }

  /// Evaluates spending against threshold and triggers local notification if exceeded
  static Future<void> evaluateThreshold(ExpenseCategory category) async {
    final budget = getBudget(category);
    if (budget <= 0.0) return; // No threshold configured

    final spent = getCurrentMonthSpent(category);
    final ratio = spent / budget;

    if (ratio >= 1.0) {
      await NotificationService.showBudgetAlert(
        id: category.index * 10 + 1,
        title: '⚠️ Budget Exceeded: ${category.label}',
        body: 'You have spent \$${spent.toStringAsFixed(2)} of your \$${budget.toStringAsFixed(2)} budget this month.',
      );
    } else if (ratio >= 0.8) {
      await NotificationService.showBudgetAlert(
        id: category.index * 10 + 2,
        title: '🔔 Budget Warning: ${category.label}',
        body: 'You have used ${(ratio * 100).toStringAsFixed(0)}% (\$${spent.toStringAsFixed(2)}/\$${budget.toStringAsFixed(2)}) of your monthly limit.',
      );
    }
  }
}
```

Implement duplicate receipt and recurring expense detection in Flutter.


Here is the complete implementation for **duplicate receipt detection** (to prevent accidental double-scanning and expense fraud) and **recurring subscription detection** (identifying fixed monthly/yearly cadence charges) in DocuMorph.

---

**Step 1: Detection Engine Service (`lib/services/duplicate_and_recurring_service.dart`)**

This service handles fuzzy vendor matching, duplicate scoring (exact match on amount, date, and invoice number), and periodicity analysis for recurring payments:

```dart
import 'storage_service.dart';
import 'structured_parser_service.dart';

class DuplicateMatch {
  final StoredDocument existingDoc;
  final double confidenceScore; // 0.0 to 1.0
  final String reason;

  DuplicateMatch({
    required this.existingDoc,
    required this.confidenceScore,
    required this.reason,
  });
}

class RecurringSubscription {
  final String vendor;
  final double amount;
  final String cadence; // 'Monthly' or 'Annual'
  final int occurrenceCount;
  final DateTime lastCharged;

  RecurringSubscription({
    required this.vendor,
    required this.amount,
    required this.cadence,
    required this.occurrenceCount,
    required this.lastCharged,
  });
}

class DuplicateAndRecurringService {
  // -------------------------------------------------------------
  // 1. DUPLICATE RECEIPT DETECTION
  // -------------------------------------------------------------
  static DuplicateMatch? checkDuplicate(ParsedDocument candidate) {
    final allDocs = StorageService.getAllDocuments();
    final receipts = allDocs.where((d) => d.structuredData != null).toList();

    for (final doc in receipts) {
      final existing = doc.structuredData!;

      // Check 1: Identical non-empty Invoice Number from same vendor
      if (candidate.invoiceNumber.isNotEmpty &&
          candidate.invoiceNumber != 'N/A' &&
          existing.invoiceNumber.isNotEmpty &&
          existing.invoiceNumber != 'N/A') {
        if (candidate.invoiceNumber.toLowerCase() == existing.invoiceNumber.toLowerCase() &&
            _isVendorSimilar(candidate.vendor, existing.vendor)) {
          return DuplicateMatch(
            existingDoc: doc,
            confidenceScore: 1.0,
            reason: 'Identical Invoice #${candidate.invoiceNumber} from ${existing.vendor}',
          );
        }
      }

      // Check 2: Exact Total Amount + Exact Date + Matching Vendor
      final sameAmount = (candidate.total - existing.total).abs() < 0.01 && candidate.total > 0;
      final sameDate = candidate.date.isNotEmpty && candidate.date == existing.date;
      final sameVendor = _isVendorSimilar(candidate.vendor, existing.vendor);

      if (sameAmount && sameDate && sameVendor) {
        return DuplicateMatch(
          existingDoc: doc,
          confidenceScore: 0.95,
          reason: 'Exact match for \$${candidate.total.toStringAsFixed(2)} on ${candidate.date} (${existing.vendor})',
        );
      }

      // Check 3: Same Vendor + Same Total within 2 days (Fuzzy Date Match)
      if (sameAmount && sameVendor && candidate.date.isNotEmpty && existing.date.isNotEmpty) {
        final d1 = DateTime.tryParse(candidate.date);
        final d2 = DateTime.tryParse(existing.date);
        if (d1 != null && d2 != null && (d1.difference(d2).inDays).abs() <= 2) {
          return DuplicateMatch(
            existingDoc: doc,
            confidenceScore: 0.75,
            reason: 'Similar charge (\$${candidate.total.toStringAsFixed(2)}) from ${existing.vendor} within 48 hours',
          );
        }
      }
    }

    return null;
  }

  // -------------------------------------------------------------
  // 2. RECURRING SUBSCRIPTION & EXPENSE CADENCE FINDER
  // -------------------------------------------------------------
  static List<RecurringSubscription> detectRecurringExpenses() {
    final allDocs = StorageService.getAllDocuments();
    final receipts = allDocs.where((d) => d.structuredData != null).toList();

    // Group receipts by normalized vendor
    final Map<String, List<StoredDocument>> vendorGroups = {};
    for (final doc in receipts) {
      final vendorKey = doc.structuredData!.vendor.trim().toLowerCase();
      if (vendorKey.isEmpty) continue;
      vendorGroups.putIfAbsent(vendorKey, () => []).add(doc);
    }

    final List<RecurringSubscription> recurringList = [];

    vendorGroups.forEach((vendor, docs) {
      if (docs.length < 2) return;

      // Sort chronological
      docs.sort((a, b) {
        final d1 = DateTime.tryParse(a.structuredData!.date) ?? a.createdAt;
        final d2 = DateTime.tryParse(b.structuredData!.date) ?? b.createdAt;
        return d1.compareTo(d2);
      });

      // Group by amount
      final Map<double, List<DateTime>> amountDates = {};
      for (final doc in docs) {
        final amount = doc.structuredData!.total;
        final date = DateTime.tryParse(doc.structuredData!.date) ?? doc.createdAt;
        amountDates.putIfAbsent(amount, () => []).add(date);
      }

      amountDates.forEach((amount, dates) {
        if (dates.length >= 2 && amount > 0) {
          // Check intervals between consecutive dates
          List<int> intervals = [];
          for (int i = 0; i < dates.length - 1; i++) {
            intervals.add(dates[i + 1].difference(dates[i]).inDays);
          }

          // Evaluate monthly cadence (25-35 days interval)
          final isMonthly = intervals.every((days) => days >= 25 && days <= 35);
          // Evaluate annual cadence (350-380 days interval)
          final isAnnual = intervals.every((days) => days >= 350 && days <= 380);

          if (isMonthly || isAnnual) {
            recurringList.add(RecurringSubscription(
              vendor: docs.first.structuredData!.vendor,
              amount: amount,
              cadence: isMonthly ? 'Monthly' : 'Annual',
              occurrenceCount: dates.length,
              lastCharged: dates.last,
            ));
          }
        }
      });
    });

    return recurringList;
  }

  static bool _isVendorSimilar(String v1, String v2) {
    final s1 = v1.trim().toLowerCase();
    final s2 = v2.trim().toLowerCase();
    if (s1.isEmpty || s2.isEmpty) return false;
    return s1 == s2 || s1.contains(s2) || s2.contains(s1);
  }
}
```

### Step 3: Biometric Authentication Service (`lib/services/auth_service.dart`)

This service manages checking biometric availability, handling platform fallback to device passcode/PIN, and persisting the user's security lock toggle in Hive.

```dart
import 'package:flutter/services.dart';
import 'package:hive_flutter/hive_flutter.dart';
import 'package:local_auth/local_auth.dart';

class AuthService {
  static final LocalAuthentication _auth = LocalAuthentication();
  static const String _settingsBoxName = 'app_settings_box';
  static const String _biometricsKey = 'is_biometric_lock_enabled';

  static Future<void> init() async {
    await Hive.openBox(_settingsBoxName);
  }

  static Box _getSettingsBox() => Hive.box(_settingsBoxName);

  /// Check if biometric lock is turned on by the user in settings
  static bool isBiometricLockEnabled() {
    return _getSettingsBox().get(_biometricsKey, defaultValue: false);
  }

  /// Toggle biometric lock setting
  static Future<void> setBiometricLockEnabled(bool enabled) async {
    await _getSettingsBox().put(_biometricsKey, enabled);
  }

  /// Check if the hardware supports biometrics
  static Future<bool> canCheckBiometrics() async {
    try {
      final bool canAuthenticateWithBiometrics = await _auth.canCheckBiometrics;
      final bool canAuthenticate = canAuthenticateWithBiometrics || await _auth.isDeviceSupported();
      return canAuthenticate;
    } on PlatformException catch (_) {
      return false;
    }
  }

  /// Trigger biometric prompt with automatic OS passcode fallback
  static Future<bool> authenticate({
    String reason = 'Authenticate to access financial records and saved scans',
  }) async {
    try {
      // If user hasn't enabled the lock in settings, bypass authentication
      if (!isBiometricLockEnabled()) {
        return true;
      }

      final bool isHardwareAvailable = await canCheckBiometrics();
      if (!isHardwareAvailable) {
        return true; // Gracefully bypass if hardware does not support biometrics
      }

      return await _auth.authenticate(
        localizedReason: reason,
        options: const AuthenticationOptions(
          stickyAuth: true,
          biometricOnly: false, // Allows device PIN/Passcode fallback
          useErrorDialogs: true,
        ),
      );
    } on PlatformException catch (_) {
      return false;
    }
  }
}
```

Implement app lifecycle state listener to automatically re-lock biometric security when Flutter app is backgrounded.


Here is the complete implementation for an **App Lifecycle Security Listener** in DocuMorph. It tracks when the app is minimized or backgrounded, checks elapsed time against a configurable timeout threshold (e.g., 60 seconds), and automatically presents a secure lock screen when the user resumes.

### Step 1: Create the App Lock State Controller (`lib/services/app_lock_manager.dart`)

This singleton manager tracks background timestamps and exposes a reactive `ValueNotifier<bool>` so the entire application can transition to a locked state dynamically without disrupting the navigation stack.

```dart
import 'package:flutter/foundation.dart';
import 'auth_service.dart';

class AppLockManager {
  static final AppLockManager _instance = AppLockManager._internal();
  factory AppLockManager() => _instance;
  AppLockManager._internal();

  /// Reactive state for whether the app is currently showing the lock screen
  final ValueNotifier<bool> isLockedNotifier = ValueNotifier<bool>(false);

  DateTime? _backgroundTimestamp;

  /// Grace period before triggering an auto-lock (e.g., 60 seconds)
  static const Duration lockTimeout = Duration(seconds: 60);

  void handleAppPaused() {
    // Only track timeout if user has biometric lock enabled in settings
    if (AuthService.isBiometricLockEnabled()) {
      _backgroundTimestamp = DateTime.now();
    }
  }

  void handleAppResumed() {
    if (!AuthService.isBiometricLockEnabled()) {
      isLockedNotifier.value = false;
      _backgroundTimestamp = null;
      return;
    }

    if (_backgroundTimestamp != null) {
      final elapsed = DateTime.now().difference(_backgroundTimestamp!);
      if (elapsed >= lockTimeout) {
        // Lock threshold exceeded
        isLockedNotifier.value = true;
      }
      _backgroundTimestamp = null;
    }
  }

  /// Manually trigger immediate lock
  void lockImmediately() {
    if (AuthService.isBiometricLockEnabled()) {
      isLockedNotifier.value = true;
    }
  }

  /// Mark session as authenticated
  void unlock() {
    isLockedNotifier.value = false;
    _backgroundTimestamp = null;
  }
}

```

### Step 3: Subscription & Entitlement Service (`lib/services/subscription_service.dart`)

This service initializes RevenueCat, caches entitlement status (`pro`), fetches available store offerings (monthly and annual packages), and executes purchases and restores:

```dart
import 'dart:io';
import 'package:flutter/foundation.dart';
import 'package:flutter/services.dart';
import 'package:purchases_flutter/purchases_flutter.dart';

class SubscriptionService {
  static const String entitlementPro = 'pro';

  // Replace with your public RevenueCat API keys from your dashboard
  static const String _appleApiKey = 'appl_your_ios_public_sdk_key';
  static const String _googleApiKey = 'goog_your_android_public_sdk_key';

  static Future<void> init() async {
    await Purchases.setLogLevel(LogLevel.debug);

    PurchasesConfiguration configuration;
    if (Platform.isIOS) {
      configuration = PurchasesConfiguration(_appleApiKey);
    } else if (Platform.isAndroid) {
      configuration = PurchasesConfiguration(_googleApiKey);
    } else {
      return;
    }

    await Purchases.configure(configuration);
  }

  /// Check if the user currently has an active Pro entitlement
  static Future<bool> isUserPro() async {
    try {
      CustomerInfo customerInfo = await Purchases.getCustomerInfo();
      return customerInfo.entitlements.all[entitlementPro]?.isActive ?? false;
    } catch (e) {
      debugPrint('Error checking entitlement: $e');
      return false;
    }
  }

  /// Fetch active offerings (Monthly & Annual packages) from RevenueCat
  static Future<Offering?> getProOffering() async {
    try {
      Offerings offerings = await Purchases.getOfferings();
      if (offerings.current != null && offerings.current!.availablePackages.isNotEmpty) {
        return offerings.current;
      }
    } catch (e) {
      debugPrint('Error fetching offerings: $e');
    }
    return null;
  }

  /// Purchase a specific package (Monthly or Annual)
  static Future<bool> purchasePackage(Package package) async {
    try {
      CustomerInfo customerInfo = await Purchases.purchasePackage(package);
      return customerInfo.entitlements.all[entitlementPro]?.isActive ?? false;
    } on PlatformException catch (e) {
      var errorCode = PurchasesErrorHelper.getErrorCode(e);
      if (errorCode == PurchasesErrorCode.purchaseCancelledError) {
        debugPrint('User cancelled purchase.');
      } else {
        debugPrint('Purchase failed: ${e.message}');
      }
      return false;
    }
  }

  /// Restore previous purchases (Apple/Google requirement)
  static Future<bool> restorePurchases() async {
    try {
      CustomerInfo customerInfo = await Purchases.restorePurchases();
      return customerInfo.entitlements.all[entitlementPro]?.isActive ?? false;
    } catch (e) {
      debugPrint('Restore failed: $e');
      return false;
    }
  }
}
```

### Step 3: Zero-Knowledge AES-256 Encryption Service (`lib/services/encryption_service.dart`)

This service derives a 256-bit AES encryption key from a user passphrase using SHA-256 and handles AES-GCM/CBC encryption of JSON payloads and image byte streams:

```dart
import 'dart:convert';
import 'dart:typed_data';
import 'package:crypto/crypto.dart';
import 'package:encrypt/encrypt.dart' as enc;
import 'package:hive_flutter/hive_flutter.dart';

class EncryptionService {
  static const String _settingsBoxName = 'app_settings_box';
  static const String _passphraseKey = 'sync_encryption_passphrase';

  /// Check if user has established an encryption passphrase
  static bool hasPassphrase() {
    final box = Hive.box(_settingsBoxName);
    return box.containsKey(_passphraseKey);
  }

  /// Store user encryption passphrase locally
  static Future<void> savePassphrase(String passphrase) async {
    final box = Hive.box(_settingsBoxName);
    await box.put(_passphraseKey, passphrase.trim());
  }

  static String? getPassphrase() {
    final box = Hive.box(_settingsBoxName);
    return box.get(_passphraseKey);
  }

  /// Derives a 32-byte AES key from passphrase
  static enc.Key _deriveKey(String passphrase) {
    final bytes = utf8.encode(passphrase);
    final digest = sha256.convert(bytes);
    return enc.Key(Uint8List.fromList(digest.bytes));
  }

  /// Encrypt arbitrary string payload
  static String encryptString(String plainText, String passphrase) {
    final key = _deriveKey(passphrase);
    final iv = enc.IV.fromSecureRandom(16);
    final encrypter = enc.Encrypter(enc.AES(key, mode: enc.AESMode.cbc));

    final encrypted = encrypter.encrypt(plainText, iv: iv);
    // Return IV + CipherText encoded together
    return '${iv.base64}:${encrypted.base64}';
  }

  /// Decrypt cipher payload
  static String decryptString(String combinedCipher, String passphrase) {
    final parts = combinedCipher.split(':');
    if (parts.length != 2) throw Exception('Invalid encrypted format.');

    final iv = enc.IV.fromBase64(parts[0]);
    final encrypted = enc.Encrypted.fromBase64(parts[1]);
    final key = _deriveKey(passphrase);
    final encrypter = enc.Encrypter(enc.AES(key, mode: enc.AESMode.cbc));

    return encrypter.decrypt(encrypted, iv: iv);
  }

  /// Encrypt image byte arrays before cloud upload
  static Uint8List encryptBytes(Uint8List rawBytes, String passphrase) {
    final key = _deriveKey(passphrase);
    final iv = enc.IV.fromSecureRandom(16);
    final encrypter = enc.Encrypter(enc.AES(key, mode: enc.AESMode.cbc));

    final encrypted = encrypter.encryptBytes(rawBytes, iv: iv);
    // Prefix the first 16 bytes with IV
    final result = Uint8List(16 + encrypted.bytes.length);
    result.setRange(0, 16, iv.bytes);
    result.setRange(16, result.length, encrypted.bytes);
    return result;
  }

  /// Decrypt image bytes after cloud download
  static Uint8List decryptBytes(Uint8List encryptedData, String passphrase) {
    final ivBytes = encryptedData.sublist(0, 16);
    final cipherBytes = encryptedData.sublist(16);

    final iv = enc.IV(ivBytes);
    final key = _deriveKey(passphrase);
    final encrypter = enc.Encrypter(enc.AES(key, mode: enc.AESMode.cbc));

    final decryptedList = encrypter.decryptBytes(enc.Encrypted(cipherBytes), iv: iv);
    return Uint8List.fromList(decryptedList);
  }
}

```

### Step 4: Encrypted Cloud Sync Engine (`lib/services/cloud_sync_service.dart`)

This service authenticates anonymously or via Firebase Auth, iterates over local Hive records, encrypts the payload, and performs bidirectional delta syncing with Cloud Firestore and Firebase Storage:

```dart
import 'dart:convert';
import 'dart:io';
import 'dart:typed_data';
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:firebase_storage/firebase_storage.dart';
import 'package:path_provider/path_provider.dart';
import 'encryption_service.dart';
import 'storage_service.dart';

class SyncProgress {
  final int uploaded;
  final int downloaded;
  final String status;

  SyncProgress({required this.uploaded, required this.downloaded, required this.status});
}

class CloudSyncService {
  static final FirebaseAuth _auth = FirebaseAuth.instance;
  static final FirebaseFirestore _db = FirebaseFirestore.instance;
  static final FirebaseStorage _storage = FirebaseStorage.instance;

  /// Ensure active authenticated user session
  static Future<User> _getOrSignInUser() async {
    User? user = _auth.currentUser;
    if (user == null) {
      final cred = await _auth.signInAnonymously();
      user = cred.user!;
    }
    return user;
  }

  /// Runs full bidirectional encrypted cloud synchronization
  static Future<SyncProgress> runSync(String passphrase) async {
    final user = await _getOrSignInUser();
    final userDocRef = _db.collection('users').doc(user.uid).collection('documents');
    final localDocs = StorageService.getAllDocuments();

    int uploadedCount = 0;
    int downloadedCount = 0;

    // 1. UPLOAD LOCAL TO CLOUD (Encrypted)
    for (final doc in localDocs) {
      final rawJson = jsonEncode(doc.toMap());
      final encryptedPayload = EncryptionService.encryptString(rawJson, passphrase);

      String? remoteImagePath;

      // Encrypt and upload image if present
      if (doc.imagePath != null && doc.imagePath!.isNotEmpty) {
        final imgFile = File(doc.imagePath!);
        if (await imgFile.exists()) {
          final rawBytes = await imgFile.readAsBytes();
          final encryptedBytes = EncryptionService.encryptBytes(rawBytes, passphrase);

          final storageRef = _storage
              .ref()
              .child('users/${user.uid}/encrypted_images/${doc.id}.enc');
          await storageRef.putData(encryptedBytes);
          remoteImagePath = storageRef.fullPath;
        }
      }

      await userDocRef.doc(doc.id).set({
        'id': doc.id,
        'encryptedData': encryptedPayload,
        'hasImage': remoteImagePath != null,
        'updatedAt': doc.createdAt.toIso8601String(),
        'client': 'DocuMorph',
      }, SetOptions(merge: true));

      uploadedCount++;
    }

    // 2. DOWNLOAD CLOUD DELTAS TO LOCAL
    final cloudSnapshot = await userDocRef.get();
    final localDocIds = localDocs.map((d) => d.id).toSet();
    final appDir = await getApplicationDocumentsDirectory();

    for (final docSnap in cloudSnapshot.docs) {
      final data = docSnap.data();
      final docId = data['id'] as String;

      // If missing locally, decrypt and restore
      if (!localDocIds.contains(docId)) {
        final encryptedData = data['encryptedData'] as String;
        final decryptedJson = EncryptionService.decryptString(encryptedData, passphrase);
        final Map<String, dynamic> docMap = jsonDecode(decryptedJson);

        // Download & decrypt image if present
        if (data['hasImage'] == true) {
          final storageRef = _storage
              .ref()
              .child('users/${user.uid}/encrypted_images/$docId.enc');
          final Uint8List? downloadedEncBytes = await storageRef.getData();

          if (downloadedEncBytes != null) {
            final decryptedImgBytes = EncryptionService.decryptBytes(downloadedEncBytes, passphrase);
            final localImageFile = File('${appDir.path}/documorph_images/$docId.jpg');
            await localImageFile.parent.create(recursive: true);
            await localImageFile.writeAsBytes(decryptedImgBytes);
            docMap['imagePath'] = localImageFile.path;
          }
        }

        final restoredDoc = StoredDocument.fromMap(docMap);
        await StorageService.saveDocument(restoredDoc);
        downloadedCount++;
      }
    }

    return SyncProgress(
      uploaded: uploadedCount,
      downloaded: downloadedCount,
      status: 'Sync completed successfully.',
    );
  }
}
```

### Step 5: Encrypted Cloud Sync Modal UI (`lib/widgets/cloud_sync_dialog.dart`)

This dialog prompts the user for their encryption passphrase, displays real-time sync progress, and prevents plaintext leaks:

```dart
import 'package:flutter/material.dart';
import '../services/cloud_sync_service.dart';
import '../services/encryption_service.dart';

class CloudSyncDialog extends StatefulWidget {
  final VoidCallback onSyncSuccess;

  const CloudSyncDialog({super.key, required this.onSyncSuccess});

  static Future<void> show(BuildContext context, {required VoidCallback onSyncSuccess}) {
    return showDialog(
      context: context,
      builder: (ctx) => CloudSyncDialog(onSyncSuccess: onSyncSuccess),
    );
  }

  @override
  State<CloudSyncDialog> createState() => _CloudSyncDialogState();
}

class _CloudSyncDialogState extends State<CloudSyncDialog> {
  final TextEditingController _passphraseController = TextEditingController();
  bool _isSyncing = false;
  String? _statusMessage;
  String? _errorMessage;

  @override
  void initState() {
    super.initState();
    final saved = EncryptionService.getPassphrase();
    if (saved != null) {
      _passphraseController.text = saved;
    }
  }

  Future<void> _handleSync() async {
    final passphrase = _passphraseController.text.trim();
    if (passphrase.length < 6) {
      setState(() => _errorMessage = 'Passphrase must be at least 6 characters long.');
      return;
    }

    setState(() {
      _isSyncing = true;
      _errorMessage = null;
      _statusMessage = 'Encrypting local database and connecting to Firestore...';
    });

    try {
      await EncryptionService.savePassphrase(passphrase);
      final result = await CloudSyncService.runSync(passphrase);

      if (!mounted) return;

      widget.onSyncSuccess();
      Navigator.pop(context);

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          backgroundColor: const Color(0xFF10B981),
          content: Text(
            '🔒 Sync Complete! (${result.uploaded} uploaded, ${result.downloaded} downloaded)',
          ),
        ),
      );
    } catch (e) {
      setState(() {
        _isSyncing = false;
        _errorMessage = 'Sync failed: $e';
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return AlertDialog(
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
      title: Row(
        children: const [
          Icon(Icons.cloud_sync_outlined, color: Color(0xFF4F46E5)),
          SizedBox(width: 8),
          Text('Encrypted Cloud Sync'),
        ],
      ),
      content: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text(
            'DocuMorph uses End-to-End Zero-Knowledge Encryption. Your data is encrypted with AES-256 before being sent to the cloud.',
            style: TextStyle(fontSize: 13, height: 1.4),
          ),
          const SizedBox(height: 16),
          TextField(
            controller: _passphraseController,
            obscureText: true,
            decoration: InputDecoration(
              labelText: 'Encryption Key / Passphrase',
              prefixIcon: const Icon(Icons.key),
              border: OutlineInputBorder(borderRadius: BorderRadius.circular(10)),
              hintText: 'Enter secret passphrase',
            ),
          ),
          if (_errorMessage != null) ...[
            const SizedBox(height: 8),
            Text(_errorMessage!, style: const TextStyle(color: Colors.red, fontSize: 12)),
          ],
          if (_isSyncing) ...[
            const SizedBox(height: 16),
            Center(
              child: Column(
                children: [
                  const CircularProgressIndicator(),
                  const SizedBox(height: 8),
                  Text(_statusMessage ?? '', style: const TextStyle(fontSize: 11, color: Colors.grey)),
                ],
              ),
            ),
          ],
        ],
      ),
      actions: [
        TextButton(
          onPressed: _isSyncing ? null : () => Navigator.pop(context),
          child: const Text('Cancel'),
        ),
        ElevatedButton.icon(
          icon: const Icon(Icons.lock, size: 16),
          label: const Text('Sync Now'),
          style: ElevatedButton.styleFrom(
            backgroundColor: const Color(0xFF4F46E5),
            foregroundColor: Colors.white,
          ),
          onPressed: _isSyncing ? null : _handleSync,
        ),
      ],
    );
  }
}
```
